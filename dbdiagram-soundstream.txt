Table countries {
  country_id int [pk, increment]
  name varchar(100) [not null]
  code char(2) [not null, unique] 

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table users {
  user_id int [pk, increment]
  name varchar(100) [not null]
  email varchar(150) [not null, unique]
  birth_date date [not null]
  country_id int [not null]
  registration_date datetime [not null]

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table subscription_plans {
  plan_id int [pk, increment]
  name varchar(50) [not null, unique] 
  monthly_price decimal(10,2) [not null]
  max_members int 

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table user_subscriptions {
  user_subscription_id int [pk, increment]
  user_id int [not null]
  plan_id int [not null]
  start_date date [not null]
  end_date date 
  status varchar(20) [not null] 

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table subscription_payments {
  payment_id int [pk, increment]
  user_subscription_id int [not null]
  payment_date date [not null]
  amount decimal(10,2) [not null]

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table artists {
  artist_id int [pk, increment]
  name varchar(150) [not null]
  country_id int

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table albums {
  album_id int [pk, increment]
  artist_id int [not null]
  title varchar(150) [not null]
  release_date date
  cover_url varchar(255)

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table genres {
  genre_id int [pk, increment]
  name varchar(100) [not null, unique]

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table songs {
  song_id int [pk, increment]
  album_id int [not null]
  genre_id int
  title varchar(150) [not null]
  duration_seconds int [not null] 

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}

Table listens {
  listen_id bigint [pk, increment]
  user_id int [not null]
  song_id int [not null]
  listen_datetime datetime [not null]
  seconds_played int [not null]

  created_at datetime
  created_by varchar(50)
  updated_at datetime
  updated_by varchar(50)
  active boolean
}


Ref: users.country_id > countries.country_id

Ref: user_subscriptions.user_id > users.user_id
Ref: user_subscriptions.plan_id > subscription_plans.plan_id

Ref: subscription_payments.user_subscription_id > user_subscriptions.user_subscription_id

Ref: artists.country_id > countries.country_id

Ref: albums.artist_id > artists.artist_id

Ref: songs.album_id > albums.album_id
Ref: songs.genre_id > genres.genre_id

Ref: listens.user_id > users.user_id
Ref: listens.song_id > songs.song_id
